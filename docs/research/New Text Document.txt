Code mẫu T2S_Clone.py:Pythonimport asyncio
import edge_tts
import pygame
import os
import fitz  # PyMuPDF

# 1. Hàm trích xuất văn bản từ PDF
def read_pdf(file_path):
    doc = fitz.open(file_path)
    text = ""
    for page in doc:
        text += page.get_text()
    return text

# 2. Hàm chuyển văn bản thành tiếng nói và phát
async def speak(text, voice="vi-VN-HoaiMyNeural"):
    output_file = "output.mp3"
    communicate = edge_tts.Communicate(text, voice)
    await communicate.save(output_file)

    # Phát nhạc bằng pygame
    pygame.mixer.init()
    pygame.mixer.music.load(output_file)
    pygame.mixer.music.play()
    while pygame.mixer.music.get_busy():
        continue
    pygame.mixer.quit()

if __name__ == "__main__":
    content = "Chào bạn, đây là ứng dụng T2S phiên bản PC tự chế."
    # Bạn có thể thay content bằng read_pdf("document.pdf")
    asyncio.run(speak(content))
